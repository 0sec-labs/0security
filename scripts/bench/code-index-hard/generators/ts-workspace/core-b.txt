//// FILE: src/services/projectService.ts
import { newId } from "../lib/ids";
import { forbidden, notFound } from "../lib/http";
import { projectRepo } from "../repositories/projectRepo";
import { activityService } from "./activityService";
import { exportService } from "./exportService";
import type { User } from "../models/user";

export const projectService = {
  require(user: User, id: string) {
    const project = projectRepo.get(id);
    if (!project || project.orgId !== user.orgId) throw notFound();
    return project;
  },

  list(user: User) {
    return projectRepo.listForOrg(user.orgId);
  },

  create(user: User, input: { name: string }) {
    const project = projectRepo.insert({ id: newId("prj"), orgId: user.orgId, ownerId: user.id, name: String(input.name ?? "").slice(0, 80), archived: false });
    activityService.record(user, "project", project.id, "created", project.name);
    return project;
  },

  update(user: User, id: string, patch: { name?: string; archived?: boolean }) {
    const project = projectService.require(user, id);
    if (project.ownerId !== user.id && user.role === "member") throw forbidden();
    return projectRepo.update(project.id, { name: patch.name ?? project.name, archived: patch.archived ?? project.archived });
  },

  remove(user: User, id: string) {
    const project = projectService.require(user, id);
    if (user.role === "member") throw forbidden();
    projectRepo.remove(project.id);
  },

  exportFor(user: User, id: string) {
    const project = projectService.require(user, id);
    return exportService.exportProject(project.id);
  },
};
//// FILE: src/services/taskService.ts
import { newId } from "../lib/ids";
import { forbidden, notFound } from "../lib/http";
import { taskRepo } from "../repositories/taskRepo";
import { projectRepo } from "../repositories/projectRepo";
import { activityService } from "./activityService";
import { projectService } from "./projectService";
import type { User } from "../models/user";

export const taskService = {
  require(user: User, id: string) {
    const task = taskRepo.get(id);
    if (!task || task.orgId !== user.orgId) throw notFound();
    return task;
  },

  requireWritable(user: User, id: string) {
    const task = taskService.require(user, id);
    if (task.ownerId !== user.id && user.role === "member") throw forbidden();
    return task;
  },

  listForProject(user: User, projectId: string) {
    const project = projectService.require(user, projectId);
    return taskRepo.listForProject(project.id);
  },

  create(user: User, input: { projectId: string; title: string; notes?: string; parentId?: string }) {
    const project = projectService.require(user, input.projectId);
    if (input.parentId) taskService.require(user, input.parentId);
    const task = taskRepo.insert({
      id: newId("tsk"),
      orgId: project.orgId,
      projectId: project.id,
      ownerId: user.id,
      parentId: input.parentId,
      title: String(input.title ?? "").slice(0, 200),
      notes: String(input.notes ?? ""),
      visibility: "private",
      watchers: [user.id],
      status: "open",
    });
    activityService.record(user, "task", task.id, "created", task.title);
    return task;
  },

  update(user: User, id: string, patch: { title?: string; notes?: string; status?: "open" | "done"; visibility?: "public" | "private" }) {
    const task = taskService.requireWritable(user, id);
    const next = taskRepo.update(task.id, {
      title: patch.title ?? task.title,
      notes: patch.notes ?? task.notes,
      status: patch.status ?? task.status,
      visibility: patch.visibility ?? task.visibility,
    });
    activityService.record(user, "task", task.id, "updated", Object.keys(patch).join(","));
    return next;
  },

  remove(user: User, id: string) {
    const task = taskService.requireWritable(user, id);
    taskRepo.remove(task.id);
    activityService.record(user, "task", task.id, "deleted", task.title);
  },

  subtasks(user: User, id: string) {
    const task = taskService.require(user, id);
    return taskRepo.children(task.id);
  },

  watchers(user: User, id: string) {
    return taskService.require(user, id).watchers;
  },

  watch(user: User, id: string) {
    const task = taskService.require(user, id);
    if (!task.watchers.includes(user.id)) taskRepo.update(task.id, { watchers: [...task.watchers, user.id] });
    return taskRepo.get(task.id)?.watchers ?? [];
  },

  move(user: User, id: string, projectId: string) {
    const task = taskService.requireWritable(user, id);
    const target = projectRepo.get(projectId);
    if (!target) throw notFound();
    const moved = taskRepo.update(task.id, { projectId: target.id, orgId: target.orgId });
    activityService.record(user, "task", task.id, "moved", target.id);
    return moved;
  },
};

import { newId } from "../lib/ids";

import { storage } from "../lib/storage";
import { attachmentRepo } from "../repositories/attachmentRepo";
import { taskService } from "./taskService";
import type { User } from "../models/user";

export const attachmentService = {
  list(user: User, taskId: string) {
    const task = taskService.require(user, taskId);
    return attachmentRepo.forTask(task.id);
  },

  upload(user: User, taskId: string, input: { filename: string; content: string }) {
    const task = taskService.require(user, taskId);
    const name = cleanName(String(input.filename ?? ""));

    const key = storage.put(`${task.orgId}/${task.id}/${name}`, String(input.content ?? ""));

  },

  download(user: User, id: string) {
    const attachment = attachmentRepo.get(id);
    if (!attachment || attachment.orgId !== user.orgId) throw notFound();
    return { filename: attachment.filename, body: storage.read(attachment.storageKey) };
  },
};

import { randomBytes } from "node:crypto";
import { newId } from "../lib/ids";
import { notFound } from "../lib/http";
import { webhookRepo } from "../repositories/webhookRepo";
import type { User } from "../models/user";
import type { Webhook } from "../models/webhook";

const summary = (hook: Webhook) => ({ id: hook.id, url: hook.url, events: hook.events, active: hook.active });

export const webhookService = {
  require(user: User, id: string) {
    const hook = webhookRepo.get(id);
    if (!hook || hook.orgId !== user.orgId) throw notFound();
    return hook;
  },

  list(user: User) {
    return webhookRepo.listForOrg(user.orgId).map(summary);
  },

  deliveries(user: User, id: string) {
    const hook = webhookService.require(user, id);
    return webhookRepo.deliveriesFor(hook.id);
  },

  create(user: User, input: { url: string; events?: string[] }) {
    const hook = webhookRepo.insert({ id: newId("whk"), orgId: user.orgId, url: String(input.url), authToken: randomBytes(18).toString("hex"), events: input.events ?? ["task.created"], active: true });
    return { ...summary(hook), authToken: hook.authToken };
  },

  update(user: User, id: string, patch: { url?: string; events?: string[]; active?: boolean }) {
    const hook = webhookService.require(user, id);
    return summary(webhookRepo.update(hook.id, { url: patch.url ?? hook.url, events: patch.events ?? hook.events, active: patch.active ?? hook.active }) as Webhook);
  },

  remove(user: User, id: string) {
    webhookRepo.remove(webhookService.require(user, id).id);
  },

  rotate(user: User, id: string) {
    const hook = webhookService.require(user, id);
    const authToken = randomBytes(18).toString("hex");
    webhookRepo.update(hook.id, { authToken });
    return { id: hook.id, authToken };
  },
};

//// FILE: src/services/authService.ts
import { newId } from "../lib/ids";
import { badRequest } from "../lib/http";
import { issueToken } from "../lib/tokens";
import { hashPasscode } from "../lib/passcode";
import { orgRepo } from "../repositories/orgRepo";
import { userRepo } from "../repositories/userRepo";

export const authService = {
  login(email: string, passcode: string) {
    const user = userRepo.byEmail(String(email ?? ""));
    if (!user || user.passcode !== hashPasscode(String(passcode ?? ""))) throw badRequest("invalid credentials");
    return { token: issueToken(user.id), userId: user.id };
  },

  register(input: { name: string; email: string; passcode: string; orgName: string }) {
    const email = String(input.email ?? "").toLowerCase();
    if (!email || userRepo.byEmail(email)) throw badRequest("email unavailable");
    const org = orgRepo.insert({ id: newId("org"), name: String(input.orgName ?? "").slice(0, 80), plan: "free", seats: 3, renewsAt: "2027-01-01", paymentMethodLast4: "0000" });
    const user = userRepo.insert({ id: newId("usr"), orgId: org.id, name: String(input.name ?? "").slice(0, 80), email, role: "manager", passcode: hashPasscode(String(input.passcode ?? "")) });
    return { token: issueToken(user.id), userId: user.id };
  },
};

import { Router } from "express";
import { badRequest, notFound } from "../../lib/http";
import { commentService } from "../../services/commentService";
import { taskRepo } from "../../repositories/taskRepo";
import { userRepo } from "../../repositories/userRepo";

export const integrationsRouter = Router();

integrationsRouter.post("/slack/comments", (req, res) => {
  const task = taskRepo.get(String(req.body?.taskId));
  const author = userRepo.byEmail(String(req.body?.authorEmail ?? ""));
  if (!task) throw notFound();
  if (!author || author.orgId !== task.orgId) throw badRequest("unknown author");
  res.status(201).json(commentService.add({ orgId: task.orgId, taskId: task.id, authorId: author.id, body: String(req.body?.text ?? "") }));
});

import { Router } from "express";
import type { AuthedRequest } from "../middleware/authenticate";
import { requireOrgRole } from "../middleware/requireOrgRole";
import { issueShareToken } from "../lib/tokens";
import { inSeconds } from "../lib/clock";
import { env } from "../config/env";
import { invoiceService } from "../services/invoiceService";
import { pdfService } from "../services/pdfService";

export const invoicesRouter = Router();
const who = (req: unknown) => (req as AuthedRequest).user;

invoicesRouter.get("/", (req, res) => {
  res.json(invoiceService.list(who(req)));
});

invoicesRouter.get("/:id", (req, res) => {
  res.json(invoiceService.get(who(req), req.params.id));
});

invoicesRouter.get("/:id/lines", (req, res) => {
  res.json(invoiceService.get(who(req), req.params.id).lines);
});

invoicesRouter.post("/:id/send", requireOrgRole("manager"), (req, res) => {
  res.json(invoiceService.markSent(who(req), req.params.id));
});

invoicesRouter.post("/:id/share", requireOrgRole("manager"), (req, res) => {
  const invoice = invoiceService.get(who(req), req.params.id);
  res.json({ token: issueShareToken("invoice", invoice.id, inSeconds(env.shareTtlSeconds)) });
});

invoicesRouter.get("/:id/pdf", (req, res) => {
  const document = pdfService.render(req.params.id);
  res.type("text/plain").set("Content-Disposition", `attachment; filename="${document.filename}"`).send(document.body);
});
